void dijkstra(int start) {
    memset(dis, 0x3f, sizeof dis);
    int init_mask = key[start];
    dis[init_mask][start] = 0;
    q.push(make_pair(0, make_pair(start, init_mask)));

    while (!q.empty()) {
        int d = q.top().first;
        int u = q.top().second.first;
        int mask = q.top().second.second;
        q.pop();

        if (d &gt; dis[mask][u]) continue;

        for (size_t i = 0; i &lt; g[u].size(); ++i) {
            int v = g[u][i].v;
            int w = g[u][i].w;
            int need = g[u][i].need;

            // 状态不满足通行条件，跳过
            if ((mask &amp; need) != need) continue;

            // 到达v后，合并v的状态（拾取钥匙）
            int next_mask = mask | key[v];
            if (dis[next_mask][v] &gt; d + w) {
                dis[next_mask][v] = d + w;
                q.push(make_pair(dis[next_mask][v], make_pair(v, next_mask)));
            }
        }
    }

bool spfa(int s){
    for(int i=1;i&lt;=n;i++){
        dis[i]=INF;
        inq[i]=false;
        cnt[i]=0;
    }
    queue&lt;int&gt; q;
    dis[s]=0;
    q.push(s);
    inq[s]=true;
    cnt[s]++;

    while(!q.empty()){
        int u=q.front();
        q.pop();
        inq[u]=false;

        for(auto &amp;e:g[u]){
            int v=e.v;
            ll w=e.w;
            if(dis[v] &gt; dis[u] + w){
                dis[v] = dis[u]+w;
                if(!inq[v]){
                    q.push(v);
                    inq[v]=true;
                    cnt[v]++;
                    if(cnt[v]&gt;=n){
                        return false; //存在负环，从s可达
                    }
                }
            }
        }
    }
    return true;
